65CartesianPosition canonicalize(const CartesianPosition& point, units::Angle azimuth)
66{
67 using namespace boost::math::double_constants;
68 // area.angle is azimuth angle of EN 302 931 V1.1.1
69 const units::Angle zenith = half_pi * units::si::radian - azimuth;
70 const double sin_z = sin(zenith);
71 const double cos_z = cos(zenith);
72 // rotate canonical point around origin clockwise: zenith = 90 deg - azimuth
73 // other interpretation: rotate shape's long side onto abscissa
74 CartesianPosition canonical;
75 canonical.x = cos_z * point.x + sin_z * point.y;
76 canonical.y = -sin_z * point.x + cos_z * point.y;
77 return canonical;
78}
