52bool DefaultLocationChecker::check_identified_region(const PositionFix& location, const Vanetza_Security_SequenceOfIdentifiedRegion& seq) const
53{
54 if (!country_db_ || country_db_->empty())

58 }
59
60 // reject any region by default

62

66 continue;
67 }

74 }
75 } else {
