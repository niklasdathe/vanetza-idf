24ecdsa256::KeyPair load_private_key_from_file(const std::string& key_path)
25{
26 const PrivateKey private_key = load_private_key_from_der_file(key_path);
27 if (private_key.type != KeyType::NistP256) {
28 // ETSI TS 103 097 v1.2.1 only defines NIST P-256 keys for v2 security.
29 throw std::runtime_error("v2 private key must use the NIST P-256 curve: " + key_path);
30 }
31
32#if defined(VANETZA_WITH_OPENSSL)
33 const security::PublicKey public_key = openssl::derive_public_key(private_key);
34#elif defined(VANETZA_WITH_CRYPTOPP)
35 const security::PublicKey public_key = cryptopp::derive_public_key(private_key);
36#else
37# warning "no crypto backend available for v2 private key loading"
38 const security::PublicKey public_key;
39#endif
40
41 ecdsa256::KeyPair key_pair;
42 std::copy(private_key.key.begin(), private_key.key.end(), key_pair.private_key.key.begin());
43 std::copy(public_key.x.begin(), public_key.x.end(), key_pair.public_key.x.begin());
44 std::copy(public_key.y.begin(), public_key.y.end(), key_pair.public_key.y.begin());
45 return key_pair;
46}

48bool save_private_key_pkcs8_der(std::ostream& os, const ecdsa256::KeyPair& key_pair)
49{
50 // PKCS#8 PrivateKeyInfo wrapping SEC 1 ECPrivateKey for secp256r1
51 static const uint8_t header[] = {
52 0x30, 0x81, 0x87, /*< SEQUENCE, length 135 */
53 0x02, 0x01, 0x00, /*< INTEGER 0 (version) */
54 0x30, 0x13, /*< SEQUENCE (AlgorithmIdentifier) */
55 0x06, 0x07, 0x2a, 0x86, 0x48, 0xce, 0x3d, 0x02, 0x01, /*< OID 1.2.840.10045.2.1 (ecPublicKey) */
56 0x06, 0x08, 0x2a, 0x86, 0x48, 0xce, 0x3d, 0x03, 0x01, 0x07, /*< OID 1.2.840.10045.3.1.7 (secp256r1) */
57 0x04, 0x6d, /*< OCTET STRING, length 109 */
58 0x30, 0x6b, /*< SEQUENCE (ECPrivateKey), length 107 */
59 0x02, 0x01, 0x01, /*< INTEGER 1 (version) */
60 0x04, 0x20, /*< OCTET STRING, length 32 */
61 };
62 static const uint8_t pub_header[] = {
63 0xa1, 0x44, /*< [1] CONSTRUCTED, length 68 */
64 0x03, 0x42, /*< BIT STRING, length 66 */
65 0x00, /*< 0 unused bits */
66 0x04, /*< uncompressed point (x, y) */
67 };
68
69 os.write(reinterpret_cast<const char*>(header), sizeof(header));
70 os.write(reinterpret_cast<const char*>(key_pair.private_key.key.data()), key_pair.private_key.key.size());
71 os.write(reinterpret_cast<const char*>(pub_header), sizeof(pub_header));
72 os.write(reinterpret_cast<const char*>(key_pair.public_key.x.data()), key_pair.public_key.x.size());
73 os.write(reinterpret_cast<const char*>(key_pair.public_key.y.data()), key_pair.public_key.y.size());
74 return os.good();
75}
